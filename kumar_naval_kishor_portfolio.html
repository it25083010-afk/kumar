# Python Calculator by Kumar Naval Kishor
import tkinter as tk

class CalculatorApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Calculator - Kumar Naval Kishor")
        self.expression = ""
        self.display_var = tk.StringVar(value="0")

        entry = tk.Entry(root, textvariable=self.display_var,
                         font=("Consolas", 20), justify="right")
        entry.grid(row=0, column=0, columnspan=4, padx=10, pady=10)

        buttons = [
            ('7',1,0), ('8',1,1), ('9',1,2), ('/',1,3),
            ('4',2,0), ('5',2,1), ('6',2,2), ('*',2,3),
            ('1',3,0), ('2',3,1), ('3',3,2), ('-',3,3),
            ('C',4,0), ('0',4,1), ('=',4,2), ('+',4,3),
        ]
        for (text, r, c) in buttons:
            tk.Button(root, text=text, width=5, height=2,
                      command=lambda t=text: self.on_press(t)).grid(row=r, column=c)

    def on_press(self, char):
        if char == 'C':
            self.expression = ""
            self.display_var.set("0")
        elif char == '=':
            try:
                result = str(eval(self.expression))
                self.display_var.set(result)
                self.expression = result
            except ZeroDivisionError:
                self.display_var.set("Error: Div by 0")
                self.expression = ""
        else:
            self.expression += char
            self.display_var.set(self.expression)

if __name__ == "__main__":
    root = tk.Tk()
    CalculatorApp(root)
    root.mainloop()

// Java Entry Form by Kumar Naval Kishor
import javax.swing.*;
import javax.swing.table.DefaultTableModel;
import java.awt.*;

public class EntryForm extends JFrame {
    private JTextField nameField, emailField, phoneField;
    private JComboBox<String> courseBox;
    private DefaultTableModel tableModel;

    public EntryForm() {
        setTitle("Student Entry Form - Kumar Naval Kishor");
        setSize(550, 420);
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLayout(new BorderLayout(10, 10));

        JPanel formPanel = new JPanel(new GridLayout(5, 2, 8, 8));
        formPanel.setBorder(BorderFactory.createEmptyBorder(12, 12, 12, 12));

        nameField = new JTextField();
        emailField = new JTextField();
        phoneField = new JTextField();
        courseBox = new JComboBox<>(new String[]{
            "Web Development", "Java Programming", "Python & AI", "UI/UX Design"
        });

        formPanel.add(new JLabel("Full Name:"));   formPanel.add(nameField);
        formPanel.add(new JLabel("Email:"));       formPanel.add(emailField);
        formPanel.add(new JLabel("Course:"));      formPanel.add(courseBox);
        formPanel.add(new JLabel("Phone:"));       formPanel.add(phoneField);

        JButton submitBtn = new JButton("Submit Entry");
        submitBtn.addActionListener(e -> addEntry());
        formPanel.add(new JLabel());               formPanel.add(submitBtn);

        tableModel = new DefaultTableModel(new String[]{"Name", "Course", "Email", "Phone"}, 0);
        add(formPanel, BorderLayout.NORTH);
        add(new JScrollPane(new JTable(tableModel)), BorderLayout.CENTER);
    }

    private void addEntry() {
        String name = nameField.getText().trim();
        String email = emailField.getText().trim();
        String phone = phoneField.getText().trim();
        if (name.isEmpty() || email.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please fill all required fields!");
            return;
        }
        tableModel.addRow(new Object[]{name, courseBox.getSelectedItem(), email, phone});
        nameField.setText(""); emailField.setText(""); phoneField.setText("");
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new EntryForm().setVisible(true));
    }
}
